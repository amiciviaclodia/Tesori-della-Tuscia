  function normalizeAngle(deg){

    return ((deg % 360) + 360) % 360;

  }


  function shortestAngleDelta(from, to){

    return ((to - from + 540) % 360) - 180;

  }


  function smoothAngle(current, target, factor = 0.28){

    if(current === null || Number.isNaN(current)) return normalizeAngle(target);

    const delta = shortestAngleDelta(current, target);

    return normalizeAngle(current + delta * factor);

  }


  function getScreenAngle(){

    if(window.screen?.orientation?.angle != null) return window.screen.orientation.angle;

    if(typeof window.orientation === 'number') return window.orientation;

    return 0;

  }


  function extractCompassHeading(evt){

    if(typeof evt.webkitCompassHeading === 'number' && !Number.isNaN(evt.webkitCompassHeading)){

      return normalizeAngle(evt.webkitCompassHeading);

    }

    if(typeof evt.alpha === 'number' && !Number.isNaN(evt.alpha)){

      return normalizeAngle(360 - evt.alpha + getScreenAngle());

    }

    return null;

  }


  function updateGeoUI(clue){

    if(!lastPosition){

      distanceValue.textContent = '— m';

      arrow.style.transform = 'rotate(0deg)';

      return;

    }


    const dist = Math.round(distanceMeters(lastPosition.lat, lastPosition.lng, clue.lat, clue.lng));

    distanceValue.textContent = dist + ' m';


    const bearing = bearingDegrees(lastPosition.lat, lastPosition.lng, clue.lat, clue.lng);

    const heading = typeof compassHeading === 'number' ? compassHeading : 0;

    const targetRelative = normalizeAngle(bearing - heading);


    const factor = dist <= 12 ? 0.22 : dist <= 25 ? 0.42 : 0.62;

    smoothedHeading = smoothAngle(smoothedHeading, targetRelative, factor);

    arrow.style.transform = `rotate(${smoothedHeading}deg)`;

  }


  function distanceMeters(lat1, lon1, lat2, lon2){

    const R = 6371000;

    const toRad = d => d * Math.PI / 180;

    const dLat = toRad(lat2 - lat1);

    const dLon = toRad(lon2 - lon1);

    const a = Math.sin(dLat/2) ** 2 + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon/2) ** 2;

    return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

  }


  function bearingDegrees(lat1, lon1, lat2, lon2){

    const toRad = d => d * Math.PI / 180;

    const toDeg = r => r * 180 / Math.PI;

    const y = Math.sin(toRad(lon2-lon1)) * Math.cos(toRad(lat2));

    const x = Math.cos(toRad(lat1))*Math.sin(toRad(lat2)) - Math.sin(toRad(lat1))*Math.cos(toRad(lat2))*Math.cos(toRad(lon2-lon1));

    return (toDeg(Math.atan2(y, x)) + 360) % 360;

  }

  async function enableCompassIfNeeded(){

    if(!window.DeviceOrientationEvent || compassStarted) return;

    compassStarted = true;


    if(typeof DeviceOrientationEvent.requestPermission === 'function'){

      try{

        const response = await DeviceOrientationEvent.requestPermission();

        if(response !== 'granted'){

          alert('Permesso bussola non concesso. La freccia userà solo il GPS e sarà meno precisa.');

          return;

        }

      }catch(err){

        alert('Permesso bussola non disponibile: ' + err.message);

        return;

      }

    }


    const handleOrientation = evt => {

      const rawHeading = extractCompassHeading(evt);

      if(rawHeading === null) return;


      const factor = 0.72;

      compassHeading = smoothAngle(compassHeading, rawHeading, factor);


      if(currentTeamPayload?.profile?.mode === 'extraurbano'){

        const clue = currentTeamPayload.profile.clues[currentTeamPayload.team.clueIndex];

        if(clue) updateGeoUI(clue);

      }

    };


    window.addEventListener('deviceorientationabsolute', handleOrientation, true);

    window.addEventListener('deviceorientation', handleOrientation, true);

  }


  async function startGeoWatch(){

    if(!navigator.geolocation) return alert('Geolocalizzazione non supportata dal dispositivo.');


    await enableCompassIfNeeded();


    if(geoWatchId !== null){

      navigator.geolocation.clearWatch(geoWatchId);

    }


    geoWatchId = navigator.geolocation.watchPosition(async pos => {

      const incoming = { lat: pos.coords.latitude, lng: pos.coords.longitude };


      if(!lastPosition){

        lastPosition = incoming;

      } else {

        lastPosition = {

          lat: lastPosition.lat * 0.35 + incoming.lat * 0.65,

          lng: lastPosition.lng * 0.35 + incoming.lng * 0.65

        };

      }


      if(

        typeof pos.coords.heading === 'number' &&

        !Number.isNaN(pos.coords.heading) &&

        typeof pos.coords.speed === 'number' &&

        pos.coords.speed > 1.5

      ){

        compassHeading = smoothAngle(compassHeading, normalizeAngle(pos.coords.heading), 0.35);

      }


      if(currentTeamPayload && currentTeamPayload.profile && currentTeamPayload.profile.mode === 'extraurbano'){

        const clue = currentTeamPayload.profile.clues[currentTeamPayload.team.clueIndex];

        if(clue){

          updateGeoUI(clue);

          const dist = distanceMeters(lastPosition.lat, lastPosition.lng, clue.lat, clue.lng);

          const now = Date.now();

          if(dist <= 10 && now - lastGeoCheckAt > 1400){

            lastGeoCheckAt = now;

            await geoCheckNow();

          }

        }

      }

    }, err => {

      alert('Geolocalizzazione non disponibile: ' + err.message);

    }, {

      enableHighAccuracy:true,

      maximumAge:0,

      timeout:4000

    });

  }


  function renderAdminSummary(state){

    if(!adminUnlocked){

      adminLoginBox.classList.remove('hidden');

      adminPanel.classList.add('hidden');

      return;

    }


    adminLoginBox.classList.add('hidden');

    adminPanel.classList.remove('hidden');


    if(state.game.status === 'waiting') gameStateMsg.textContent = 'Partita in attesa.';

    if(state.game.status === 'started') gameStateMsg.textContent = 'Partita avviata. Progressione automatica attiva.';

    if(state.game.status === 'ended' && state.game.winnerTeamId){

      const winner = state.teams.find(t => t.id === state.game.winnerTeamId);

      gameStateMsg.textContent = 'Partita conclusa. Vincitore: ' + (winner ? winner.teamName : state.game.winnerTeamId);

    } else if(state.game.status === 'ended') {

      gameStateMsg.textContent = 'Partita conclusa.';

    }


    teamsList.innerHTML = '';

    if(!state.teams.length){

      teamsList.innerHTML = '<div class="muted">Nessuna squadra registrata.</div>';

      return;

    }


    state.teams.forEach(team => {

      const profile = bootstrap?.profiles?.find(p => p.id === team.profileId);

      const div = document.createElement('div');

      div.className = 'team-item';

      div.innerHTML = `

        <div class="row">

          <div>

            <h4 style="margin:.1rem 0 .25rem;">${team.teamName}</h4>

            <div class="mono small muted">${team.id}</div>

          </div>

          <div style="flex:1"></div>

          <span class="pill">${humanState(team, state.game)}</span>

        </div>

        <div class="small muted">Capitano: ${team.captain} · Giocatori: ${team.players.join(', ')}</div>

        <div class="small muted">Profilo: ${profile ? profile.label : team.profileId}</div>

        <div class="small muted">Progressione: ${team.clueIndex} / ${profile ? profile.clueCount : '?'}</div>

        ${team.completedAt ? `<div class="small">Completata alle: <b>${new Date(team.completedAt).toLocaleTimeString()}</b></div>` : ''}

        ${team.lastError ? `<div class="msg bad">${team.lastError}</div>` : ''}

        <div class="row" style="margin-top:10px;">

          ${(team.status === 'waiting' || team.status === 'playing') ? `<button class="warn" onclick="pauseSingleTeam('${team.id}')">Sospendi</button>` : ''}

          ${team.status === 'paused' ? `<button class="secondary" onclick="resumeSingleTeam('${team.id}')">Riattiva</button>` : ''}

          ${team.status === 'paused' ? `<button class="danger" onclick="deleteSingleTeam('${team.id}')">Cancella squadra</button>` : ''}

          ${team.status === 'completed' ? `<button class="secondary" onclick="resetSingleTeam('${team.id}')">Reset squadra</button>` : ''}

        </div>

      `;

      teamsList.appendChild(div);

    });

  }

  async function resetSingleTeam(teamId){

    if(!confirm(`Vuoi eliminare la squadra conclusa ${teamId}?`)) return;


    try{

      await api('/api/admin/reset-team/' + encodeURIComponent(teamId), { method:'POST' });


      if(currentTeamId === teamId){

        currentTeamId = null;

        currentTeamPayload = null;

        localStorage.removeItem('tesori_current_team_v31');

        currentCode.textContent = '—';

        playerGame.classList.add('hidden');

      }


      await loadBootstrap();

    }

  }
